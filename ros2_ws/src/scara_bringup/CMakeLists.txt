cmake_minimum_required(VERSION 3.16)
project(scara_bringup)
find_package(ament_cmake REQUIRED)
find_package(gazebo REQUIRED)
add_library(scara_grasp_plugin SHARED src/scara_grasp_plugin.cpp)
target_include_directories(scara_grasp_plugin PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_libraries(scara_grasp_plugin ${GAZEBO_LIBRARIES})
target_compile_features(scara_grasp_plugin PRIVATE cxx_std_17)
install(TARGETS scara_grasp_plugin LIBRARY DESTINATION lib)
install(DIRECTORY launch config worlds models DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/move_to_pose.py scripts/check_stack.py scripts/scene_objects.py scripts/gripper_command.py scripts/pick_lift_demo.py DESTINATION lib/${PROJECT_NAME})
ament_package()
