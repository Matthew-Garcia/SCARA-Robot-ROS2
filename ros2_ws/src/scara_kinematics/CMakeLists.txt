cmake_minimum_required(VERSION 3.16)
project(scara_kinematics)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
add_library(scara_kinematics_plugin SHARED src/scara_plugin.cpp)
target_include_directories(scara_kinematics_plugin PRIVATE include)
ament_target_dependencies(scara_kinematics_plugin moveit_core moveit_msgs pluginlib rclcpp geometry_msgs)
pluginlib_export_plugin_description_file(moveit_core scara_plugin.xml)
install(TARGETS scara_kinematics_plugin LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_package()
